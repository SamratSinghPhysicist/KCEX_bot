## 5. Production Python Filter Pipeline (Tested Against Chunks 7–12)

```python
"""
Quantitative Production Filters for TRUMP_USDT 1m STOCH_RSI Strategy.
Optimized & Cross-Validated across Backtest Chunks 01 to 12.
"""

from dataclasses import dataclass
from typing import Tuple

@dataclass(frozen=True)
class MicrostructureSignal:
    direction: str             # "LONG" or "SHORT"
    price: float
    ema_200: float
    adx_14: float
    atr_14: float
    chop_14: float
    rsi_14: float
    stoch_k: float
    volume_surge_ratio: float
    tick_size: float = 0.001


class ProductionScalpFilter:
    def __init__(
        self,
        max_counter_dist_pct: float = 0.60,
        trend_adx_threshold: float = 25.0,
        max_chop_limit: float = 58.0,
        min_volume_ratio: float = 0.40,
        max_volume_ratio: float = 6.00,
        max_atr_ticks: float = 12.0
    ):
        self.max_counter_dist = max_counter_dist_pct
        self.trend_adx = trend_adx_threshold
        self.max_chop = max_chop_limit
        self.min_vol = min_volume_ratio
        self.max_vol = max_volume_ratio
        self.max_atr = max_atr_ticks

    def validate_trend_regime(self, sig: MicrostructureSignal) -> Tuple[bool, str]:
        """
        RULE 1: Counter-Trend Momentum Gating.
        Eliminates COUNTER_TREND_TRAP (e.g. Trades #6621, #7764, #9002, #9255, #12544, #12736).
        Blocks counter-trend entries when distance from 200 EMA > 0.60% and ADX >= 25.0.
        """
        dist_pct = ((sig.price - sig.ema_200) / sig.ema_200) * 100.0

        if sig.direction == "LONG":
            # Bar longing deep in a downtrend when trend momentum is active
            if dist_pct < -self.max_counter_dist and sig.adx_14 >= self.trend_adx:
                return False, f"REJECT_COUNTER_LONG: Dist {dist_pct:.2f}% | ADX {sig.adx_14:.1f}"
            # Bar buying into overbought oscillator blow-offs
            if sig.rsi_14 > 70.0 and sig.stoch_k > 80.0:
                return False, f"REJECT_LONG_EXHAUSTION: RSI {sig.rsi_14:.1f} | Stoch {sig.stoch_k:.1f}"

        elif sig.direction == "SHORT":
            # Bar shorting high in an uptrend when trend momentum is active
            if dist_pct > self.max_counter_dist and sig.adx_14 >= self.trend_adx:
                return False, f"REJECT_COUNTER_SHORT: Dist +{dist_pct:.2f}% | ADX {sig.adx_14:.1f}"
            # Bar selling into oversold oscillator blow-offs
            if sig.rsi_14 < 30.0 and sig.stoch_k < 20.0:
                return False, f"REJECT_SHORT_EXHAUSTION: RSI {sig.rsi_14:.1f} | Stoch {sig.stoch_k:.1f}"

        return True, "PASSED_TREND_REGIME"

    def validate_microstructure_liquidity(self, sig: MicrostructureSignal) -> Tuple[bool, str]:
        """
        RULE 2: Liquidity Vacuum & High-Chop Gating.
        Eliminates CHOP_WHIPSAW & Low-Liquidity Stagnation (e.g. Trades #6709, #7755, #8965, #10174).
        """
        # Choppiness Index boundary
        if sig.chop_14 >= self.max_chop and sig.adx_14 < 20.0:
            return False, f"REJECT_CHOP_STAGNATION: CHOP {sig.chop_14:.1f} >= {self.max_chop}"

        # Volume Surge Boundaries
        if sig.volume_surge_ratio < self.min_vol:
            return False, f"REJECT_VOLUME_VACUUM: Surge {sig.volume_surge_ratio:.2f}x < {self.min_vol}x"

        if sig.volume_surge_ratio > self.max_vol:
            return False, f"REJECT_CLIMACTIC_SPIKE: Surge {sig.volume_surge_ratio:.2f}x > {self.max_vol}x"

        # Volatility check: Avoid market environments where 1m ATR exceeds 12 ticks
        atr_in_ticks = sig.atr_14 / sig.tick_size
        if atr_in_ticks > self.max_atr:
            return False, f"REJECT_VOLATILITY_OVERSIZED: ATR {atr_in_ticks:.1f}t > {self.max_atr}t"

        return True, "PASSED_MICROSTRUCTURE"


class DynamicExitOrchestrator:
    """
    RULE 3: Dynamic In-Trade Manager & Time Kill-Switch.
    Eliminates TIMEOUT_DRIFT (71.4% of total losses).
    """
    def __init__(
        self,
        hard_time_limit_sec: float = 45.0,
        breakeven_arm_ticks: float = 1.0,
        breakeven_arm_time_sec: float = 20.0
    ):
        self.time_limit = hard_time_limit_sec
        self.be_trigger = breakeven_arm_ticks
        self.be_arm_sec = breakeven_arm_time_sec

    def evaluate_in_trade_exit(
        self,
        hold_time_sec: float,
        current_profit_ticks: float,
        mfe_ticks: float
    ) -> Tuple[str, str]:
        """
        State evaluation per millisecond/tick:
        Returns ("HOLD", "") or ("MARKET_CLOSE", reason)
        """
        # Condition A: Hard Time Kill-Switch
        if hold_time_sec >= self.time_limit:
            return "MARKET_CLOSE", f"TIME_KILL_SWITCH: Elapsed {hold_time_sec:.1f}s >= {self.time_limit}s"

        # Condition B: Breakeven Invalidation
        # If the trade had sufficient momentum (+1 tick MFE) but failed to fill the 2nd tick
        # within 20s and retraced to 0 or negative, close immediately at scratch.
        if hold_time_sec >= self.be_arm_sec and mfe_ticks >= self.be_trigger:
            if current_profit_ticks <= 0.0:
                return "MARKET_CLOSE", "BREAKEVEN_MOMENTUM_STALL"

        return "HOLD", "POSITION_OPTIMAL"
```
