## 4. Production-Ready Algorithmic Invalidation Pipeline

The following Python module is engineered for low-latency production execution within the bot's `FilterPipeline`. It implements:
1. **Macro-Trend Alignment (`MacroTrendAlignmentFilter`)**: Blocks counter-trend entries against extended macro structures.
2. **Microstructure Volatility & Compression (`MicrostructureVolatilityFilter`)**: Invalidates signals when ATR is too low to support tick targets or when the market is in high-consolidation chop.
3. **Adaptive Time Decay Handler (`AdaptiveTimeDecayExit`)**: Dynamically liquidates stagnant trades after 40 seconds to prevent drift-out.

```python
"""
Production Microstructure Filter Pipeline for TRUMP_USDT Scalping Engine.
Optimized for high-frequency tick capture under Zero-Fee / Low-Fee execution.
Backtest Reference: backtest_TRUMP_USDT_20260905_113427
"""

import numpy as np
from dataclasses import dataclass
from typing import Tuple, Dict, Any

@dataclass(frozen=True)
class MicrostructureConfig:
    # Asset parameters
    TICK_SIZE: float = 0.001
    TP_DELTA: float = 0.002
    
    # Filter 1: Macro-Trend Boundaries
    MAX_200_EMA_DIST_PCT: float = 0.0085      # Max 0.85% displacement from 200 EMA
    ADX_TREND_MIN: float = 20.0                # Minimum directional index for trend validation
    
    # Filter 2: Volatility & Microstructure Feasibility
    MIN_ATR_SCALING: float = 1.35             # 1m ATR must be >= 1.35x the TP delta (0.0027 USDT)
    MAX_CHOPPINESS: float = 56.5              # Strict threshold: CHOP > 56.5 indicates chop
    
    # Filter 3: Order Flow & Micro-Exhaustion
    MAX_RSI_LONG: float = 65.0                # Prevent buying overbought impulse
    MIN_RSI_SHORT: float = 35.0               # Prevent selling oversold impulse
    MIN_VOLUME_SURGE: float = 0.80            # Candle volume must be at least 0.80x of 20-period SMA
    
    # Execution: Adaptive Time Kill-Switch
    HARD_KILL_SECONDS: float = 40.0           # Cut dead trades at 40s instead of 90s


class ProductionFilterPipeline:
    def __init__(self, config: MicrostructureConfig = MicrostructureConfig()):
        self.cfg = config

    def validate_signal(
        self,
        signal_dir: str,  # 'LONG' or 'SHORT'
        current_price: float,
        fast_ema: float,
        slow_ema: float,
        ema_200: float,
        atr_14: float,
        adx_14: float,
        chop_14: float,
        rsi_14: float,
        volume_surge: float
    ) -> Tuple[bool, str]:
        """
        Evaluates active market indicators against institutional gating rules.
        Returns (is_valid: bool, rejection_code: str).
        """
        # =========================================================================
        # RULE 1: Microstructure Volatility Feasibility (Eliminates W27-W33 Bleed)
        # =========================================================================
        min_required_atr = self.cfg.TP_DELTA * self.cfg.MIN_ATR_SCALING
        if atr_14 < min_required_atr:
            return False, f"REJECT_LOW_VOLATILITY: ATR ({atr_14:.6f}) < Threshold ({min_required_atr:.6f})"

        if chop_14 > self.cfg.MAX_CHOPPINESS:
            return False, f"REJECT_CHOP_ZONE: Choppiness Index ({chop_14:.2f}) > {self.cfg.MAX_CHOPPINESS:.2f}"

        # =========================================================================
        # RULE 2: Macro Trend Alignment & Overextension (Eliminates Counter-Trend Traps)
        # =========================================================================
        distance_to_200_ema = (current_price - ema_200) / ema_200

        if signal_dir == "LONG":
            # Gating against shorting into strong bullish macro or buying falling knife
            if distance_to_200_ema < -self.cfg.MAX_200_EMA_DIST_PCT and adx_14 >= self.cfg.ADX_TREND_MIN:
                return False, f"REJECT_KNIFE_CATCH: LONG at {distance_to_200_ema*100:.2f}% below 200EMA in strong trend"
            
            # Momentum Exhaustion Filter
            if rsi_14 > self.cfg.MAX_RSI_LONG:
                return False, f"REJECT_RSI_OVERBOUGHT: RSI ({rsi_14:.2f}) > {self.cfg.MAX_RSI_LONG}"

        elif signal_dir == "SHORT":
            # Gating against shorting explosive parabolic expansion
            if distance_to_200_ema > self.cfg.MAX_200_EMA_DIST_PCT and adx_14 >= self.cfg.ADX_TREND_MIN:
                return False, f"REJECT_PARABOLIC_TRAP: SHORT at +{distance_to_200_ema*100:.2f}% above 200EMA in strong trend"
            
            # Momentum Exhaustion Filter
            if rsi_14 < self.cfg.MIN_RSI_SHORT:
                return False, f"REJECT_RSI_OVERSOLD: RSI ({rsi_14:.2f}) < {self.cfg.MIN_RSI_SHORT}"

        else:
            return False, "REJECT_INVALID_DIRECTION"

        # =========================================================================
        # RULE 3: Liquidity Support & Structural Momentum
        # =========================================================================
        if volume_surge < self.cfg.MIN_VOLUME_SURGE:
            return False, f"REJECT_ANEMIC_VOLUME: Volume Surge Ratio ({volume_surge:.2f}x) below threshold"

        # Minimum ADX floor to ensure non-stagnant regime
        if adx_14 < 16.0:
            return False, f"REJECT_ABSENT_TREND: ADX ({adx_14:.2f}) < 16.0 indicates absolute chop"

        return True, "SIGNAL_VALIDATED"


class AdaptiveExecutionController:
    """
    Manages in-trade dynamic risk state and time-decay exits.
    """
    def __init__(self, config: MicrostructureConfig = MicrostructureConfig()):
        self.cfg = config

    def evaluate_in_trade_exit(
        self,
        entry_price: float,
        current_price: float,
        direction: str,
        duration_seconds: float,
        mfe_ticks: float,
        mae_ticks: float
    ) -> Tuple[bool, str]:
        """
        Enforces execution kill-switch rules based on duration and momentum decay.
        """
        price_delta = (current_price - entry_price) if direction == "LONG" else (entry_price - current_price)
        current_ticks = price_delta / self.cfg.TICK_SIZE

        # Hard Time Kill-Switch: Cut stalled trades at 40s
        if duration_seconds >= self.cfg.HARD_KILL_SECONDS:
            if current_ticks < 1.0:
                return True, f"KILL_SWITCH_TIME_DECAY: Stagnant trade held {duration_seconds:.1f}s with delta {current_ticks:.1f} ticks"

        # Stale Trade Dynamic Invalidation (Early Exit at 25s if trade is completely flat)
        if duration_seconds >= 25.0 and mfe_ticks <= 0.0 and current_ticks <= 0.0:
            return True, f"KILL_SWITCH_MOMENTUM_ABSENT: Zero MFE after {duration_seconds:.1f}s"

        return False, "MAINTAIN_POSITION"
```
