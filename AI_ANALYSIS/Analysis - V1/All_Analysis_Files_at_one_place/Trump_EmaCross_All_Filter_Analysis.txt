## 6. 💻 Production-Ready Python Filter Rules

The following filters are designed for direct integration into the bot's execution engine (`FilterPipeline`). They run in **$O(1)$ constant time** using precalculated bar indicators.

```python
"""
TRUMP_USDT EMA_CROSSOVER Alpha Shield Filters
Optimized for 1m Microstructure & 75x Scalping Operations.
"""

from typing import Dict, Any, Tuple
import math

class MacroTrendAlignmentFilter:
    """
    Kills COUNTER_TREND_TRAP losses by verifying directional alignment 
    with the 200 EMA and filtering out overextended impulse moves.
    """
    def __init__(
        self,
        min_adx_trend: float = 24.0,
        max_counter_trend_dist_pct: float = 0.25,
        max_parabolic_extension_pct: float = 2.20
    ):
        self.min_adx = min_adx_trend
        self.max_counter_dist = max_counter_trend_dist_pct
        self.max_extension = max_parabolic_extension_pct

    def evaluate(self, signal_type: str, candle: Dict[str, Any]) -> Tuple[bool, str]:
        price = candle["close"]
        ema_200 = candle["ema_200"]
        adx = candle["adx_14"]
        
        # Distance relative to 200 EMA in percentage
        dist_to_ema200_pct = ((price - ema_200) / ema_200) * 100.0

        # Parabolic extension circuit-breaker: Do not chase extreme extensions
        if abs(dist_to_ema200_pct) >= self.max_extension:
            return False, f"PARABOLIC_EXTENSION_BLOCK: {dist_to_ema200_pct:.2f}% from 200 EMA"

        # Long Signal Filtering
        if signal_type == "LONG":
            # Disallow buying when trending downward below 200 EMA
            if dist_to_ema200_pct < -self.max_counter_dist and adx >= self.min_adx:
                return False, f"COUNTER_TREND_LONG_REJECT: {dist_to_ema200_pct:.2f}% below 200 EMA with ADX {adx:.1f}"
            
        # Short Signal Filtering
        elif signal_type == "SHORT":
            # Disallow shorting when trending upward above 200 EMA
            if dist_to_ema200_pct > self.max_counter_dist and adx >= self.min_adx:
                return False, f"COUNTER_TREND_SHORT_REJECT: +{dist_to_ema200_pct:.2f}% above 200 EMA with ADX {adx:.1f}"

        return True, "PASSED_MACRO_TREND_ALIGNMENT"


class ChopAndExpansionGateFilter:
    """
    Kills TIMEOUT_DRIFT losses by verifying market has sufficient absolute volatility
    and is not trapped in an orderbook consolidation state.
    """
    def __init__(
        self,
        min_atr_ticks: float = 2.5,
        max_chop_index: float = 56.5,
        min_bb_bandwidth: float = 0.0025,
        tick_size: float = 0.001
    ):
        self.min_atr_val = min_atr_ticks * tick_size  # e.g., 2.5 * 0.001 = 0.0025 USDT
        self.max_chop = max_chop_index
        self.min_bandwidth = min_bb_bandwidth

    def evaluate(self, signal_type: str, candle: Dict[str, Any]) -> Tuple[bool, str]:
        atr = candle["atr_14"]
        chop = candle["chop_14"]
        bb_bandwidth = candle.get("bb_bandwidth", 0.005)

        # 1. Absolute Volatility Check: Market cannot traverse spread if ATR is lower than target
        if atr < self.min_atr_val:
            return False, f"INSUFFICIENT_ATR_VOLATILITY: ATR {atr:.6f} < Minimum {self.min_atr_val:.6f}"

        # 2. Choppiness Index Barrier: Over 61.8 is extreme consolidation; >56.5 bleeds timeouts
        if chop >= self.max_chop:
            return False, f"CHOPPINESS_INDEX_TOO_HIGH: CHOP {chop:.2f} >= {self.max_chop}"

        # 3. Bollinger Bandwidth Compression (Volatility Squeeze Trap)
        if bb_bandwidth < self.min_bandwidth:
            return False, f"BB_BANDWIDTH_COMPRESSION: Bandwidth {bb_bandwidth:.5f} < {self.min_bandwidth}"

        return True, "PASSED_EXPANSION_GATE"


class RsiMomentumExhaustionFilter:
    """
    Kills MOMENTUM_FAILURE & VOLATILITY_WICK_OUT losses by preventing
    entries at exhaustion tops/bottoms.
    """
    def __init__(
        self,
        long_rsi_min: float = 46.0,
        long_rsi_max: float = 64.0,
        short_rsi_min: float = 36.0,
        short_rsi_max: float = 54.0,
        max_volume_exhaustion_ratio: float = 4.0
    ):
        self.long_min_rsi = long_rsi_min
        self.long_max_rsi = long_rsi_max
        self.short_min_rsi = short_rsi_min
        self.short_max_rsi = short_rsi_max
        self.max_vol_ratio = max_volume_exhaustion_ratio

    def evaluate(self, signal_type: str, candle: Dict[str, Any]) -> Tuple[bool, str]:
        rsi = candle["rsi_14"]
        vol_surge = candle.get("volume_surge_ratio", 1.0)

        # Blow-off volume exhaustion wick trap (e.g. 4x average volume wick reversal)
        if vol_surge >= self.max_vol_ratio:
            return False, f"VOLUME_EXHAUSTION_CLIMAX: Surge {vol_surge:.2f}x >= {self.max_vol_ratio}x"

        # Signal-specific RSI sweet-spot gates
        if signal_type == "LONG":
            if rsi > self.long_max_rsi:
                return False, f"RSI_OVERBOUGHT_EXHAUSTION: RSI {rsi:.2f} > {self.long_max_rsi}"
            if rsi < self.long_min_rsi:
                return False, f"MOMENTUM_TOO_WEAK_FOR_LONG: RSI {rsi:.2f} < {self.long_min_rsi}"

        elif signal_type == "SHORT":
            if rsi < self.short_min_rsi:
                return False, f"RSI_OVERSOLD_EXHAUSTION: RSI {rsi:.2f} < {self.short_min_rsi}"
            if rsi > self.short_max_rsi:
                return False, f"MOMENTUM_TOO_STRONG_FOR_SHORT: RSI {rsi:.2f} > {self.short_max_rsi}"

        return True, "PASSED_RSI_MOMENTUM_GATE"


class AdaptiveTimeKillSwitch:
    """
    In-trade dynamic risk management rule. Modifies exit rules based on holding duration
    and MFE progression to replace the flawed static 90-second timeout.
    """
    def __init__(
        self,
        hard_kill_seconds: float = 45.0,
        be_activation_seconds: float = 20.0,
        be_mfe_trigger_ticks: float = 1.0
    ):
        self.hard_kill_sec = hard_kill_seconds
        self.be_time_trigger = be_activation_seconds
        self.be_mfe_ticks = be_mfe_trigger_ticks

    def check_exit(
        self,
        hold_duration_seconds: float,
        current_mfe_ticks: float,
        current_pnl_ticks: float
    ) -> Tuple[bool, str]:
        # Rule 1: Break-even Scratch Activation
        # If trade reached +1 tick of MFE but is stalling after 20s, exit at scratch (0 ticks)
        if hold_duration_seconds >= self.be_time_trigger:
            if current_mfe_ticks >= self.be_mfe_ticks and current_pnl_ticks <= 0.0:
                return True, "DYNAMIC_BREAKEVEN_SCRATCH_EXIT"

        # Rule 2: Hard Time Cutoff
        # Winners hit TP in < 30s. If still open at 45s, market has stalled—kill at market.
        if hold_duration_seconds >= self.hard_kill_sec:
            return True, f"HARD_TIME_KILL_SWITCH_{self.hard_kill_sec}S"

        return False, "HOLD_ACTIVE"
```
